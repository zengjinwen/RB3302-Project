cmake_minimum_required(VERSION 3.8)
project(rb3302_amcl_motion)
find_package(ament_cmake REQUIRED)
find_package(nav2_amcl REQUIRED)
find_package(pluginlib REQUIRED)
add_library(rb3302_motion SHARED differential_motion_model.cpp)
ament_target_dependencies(rb3302_motion nav2_amcl pluginlib)
pluginlib_export_plugin_description_file(nav2_amcl plugins.xml)
install(TARGETS rb3302_motion LIBRARY DESTINATION lib)
ament_package()
